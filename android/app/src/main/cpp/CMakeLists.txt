cmake_minimum_required(VERSION 3.22)
project(ac8terrain LANGUAGES CXX)
add_library(ac8terrain SHARED renderer.cpp opengl.cpp)
add_library(ac8vulkan SHARED vulkan.cpp)
file(GLOB shader_compilers "${CMAKE_ANDROID_NDK}/shader-tools/*/glslc" "${CMAKE_ANDROID_NDK}/shader-tools/*/glslc.exe")
list(GET shader_compilers 0 glslc)
set(shader_dir "${CMAKE_CURRENT_SOURCE_DIR}/../shaders")
foreach(shader spectrum.comp terrain.vert terrain.frag)
    set(header "${CMAKE_CURRENT_BINARY_DIR}/${shader}.inc")
    add_custom_command(OUTPUT "${header}"
        COMMAND "${glslc}" -O -mfmt=c "${shader_dir}/${shader}" -o "${header}"
        DEPENDS "${shader_dir}/${shader}" "${shader_dir}/terrain.inc" VERBATIM)
    target_sources(ac8vulkan PRIVATE "${header}")
endforeach()
target_include_directories(ac8vulkan PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(ac8vulkan PRIVATE cxx_std_17)
target_link_libraries(ac8vulkan PRIVATE android log vulkan)
target_compile_features(ac8terrain PRIVATE cxx_std_17)
target_link_libraries(ac8terrain PRIVATE android log EGL GLESv3)
